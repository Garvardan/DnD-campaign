> > > ```dataviewjs
> > > // Формула КД без щита: числа, переменные из свойств и арифметика.
> > > // Без eval/new Function: неизвестные переменные и операции не исполняются.
> > > const page = dv.current();
> > > const file = app.vault.getAbstractFileByPath(page.file.path);
> > > const fm = (file && app.metadataCache.getFileCache(file)?.frontmatter) ?? page;
> > > const formula = String(fm.ac_formula ?? "10 + dex_mod").trim();
> > > const shieldBonus = Number(fm.ac_shield_bonus ?? 2);
> > > const shieldEquipped = fm.ac_shield_equipped === true ||
> > >     fm.ac_shield_equipped === "true" || fm.ac_shield_equipped === 1;
> > > function variable(name) {
> > >     const abilities = ["str", "dex", "con", "int", "wis", "cha"];
> > >     const base = name.endsWith("_mod") ? name.slice(0, -4) : "";
> > >     if (abilities.includes(base)) {
> > >         const score = Number(fm[base]);
> > >         if (fm[base] != null && fm[base] !== "" && Number.isFinite(score)) {
> > >             return Math.floor((score - 10) / 2);
> > >         }
> > >     }
> > >     if (!Object.prototype.hasOwnProperty.call(fm, name)) {
> > >         throw new Error("Неизвестная переменная: " + name);
> > >     }
> > >     const raw = fm[name];
> > >     const value = Number(raw);
> > >     if (raw === null || raw === undefined || raw === "" || !Number.isFinite(value)) {
> > >         throw new Error("Переменная " + name + " не содержит числа");
> > >     }
> > >     return value;
> > > }
> > > function calculate(source) {
> > >     // Разрешаем привычный синтаксис {dex_mod} и [dex_mod].
> > >     const expression = source
> > >         .replace(/\{([a-z_]\w*)\}/gi, "$1")
> > >         .replace(/\[([a-z_]\w*)\]/gi, "$1");
> > >     if (!expression || expression.length > 256) {
> > >         throw new Error("Заполни формулу КД (до 256 символов)");
> > >     }
> > >     const tokens = [];
> > >     const lex = /\s*(\d+(?:\.\d+)?|\.\d+|[a-z_][a-z_0-9]*|[()+\-*/%,])\s*/giy;
> > >     let offset = 0;
> > >     while (offset < expression.length) {
> > >         if (!expression.slice(offset).trim()) break;
> > >         lex.lastIndex = offset;
> > >         const found = lex.exec(expression);
> > >         if (!found) throw new Error("Недопустимый символ в формуле КД");
> > >         tokens.push(found[1]);
> > >         if (tokens.length > 120) throw new Error("Слишком длинная формула КД");
> > >         offset = lex.lastIndex;
> > >     }
> > >     let pos = 0;
> > >     const next = () => tokens[pos];
> > >     const take = token => {
> > >         if (next() !== token) throw new Error("Ожидалось «" + token + "»");
> > >         pos++;
> > >     };
> > >     function primary() {
> > >         if (next() === "(") {
> > >             pos++;
> > >             const result = sum();
> > >             take(")");
> > >             return result;
> > >         }
> > >         const token = tokens[pos++];
> > >         if (token === undefined) throw new Error("Незавершённая формула КД");
> > >         if (/^(?:\d+(?:\.\d+)?|\.\d+)$/.test(token)) return Number(token);
> > >         if (!/^[a-z_]\w*$/i.test(token)) {
> > >             throw new Error("Неожиданный символ «" + token + "»");
> > >         }
> > >         if (next() !== "(") return variable(token);
> > >         pos++;
> > >         const args = [];
> > >         if (next() !== ")") {
> > >             args.push(sum());
> > >             while (next() === ",") { pos++; args.push(sum()); }
> > >         }
> > >         take(")");
> > >         const funcs = {
> > >             floor: Math.floor, ceil: Math.ceil, round: Math.round,
> > >             abs: Math.abs, min: Math.min, max: Math.max
> > >         };
> > >         const fn = funcs[token.toLowerCase()];
> > >         if (!fn || args.length === 0 ||
> > >             ((["floor", "ceil", "round", "abs"].includes(token.toLowerCase())) && args.length !== 1)) {
> > >             throw new Error("Неизвестная функция или неверные аргументы: " + token);
> > >         }
> > >         return fn(...args);
> > >     }
> > >     function unary() {
> > >         if (next() === "+") { pos++; return unary(); }
> > >         if (next() === "-") { pos++; return -unary(); }
> > >         return primary();
> > >     }
> > >     function product() {
> > >         let value = unary();
> > >         while (["*", "/", "%"].includes(next())) {
> > >             const op = tokens[pos++], right = unary();
> > >             if ((op === "/" || op === "%") && right === 0) {
> > >                 throw new Error("Деление на ноль");
> > >             }
> > >             value = op === "*" ? value * right : op === "/" ? value / right : value % right;
> > >         }
> > >         return value;
> > >     }
> > >     function sum() {
> > >         let value = product();
> > >         while (next() === "+" || next() === "-") {
> > >             const op = tokens[pos++], right = product();
> > >             value = op === "+" ? value + right : value - right;
> > >         }
> > >         return value;
> > >     }
> > >     const result = sum();
> > >     if (pos !== tokens.length) throw new Error("Неожиданный остаток формулы КД");
> > >     if (!Number.isFinite(result)) throw new Error("Формула КД дала некорректное число");
> > >     return result;
> > > }
> > > try {
> > >     if (!Number.isSafeInteger(shieldBonus) || shieldBonus < 0 || shieldBonus > 1000) {
> > >         throw new Error("Бонус щита должен быть целым числом от 0 до 1000");
> > >     }
> > >     const base = calculate(formula);
> > >     const total = base + (shieldEquipped ? shieldBonus : 0);
> > >     const fmt = num => String(Number(num.toFixed(3)));
> > >     // Отдельное поле с вычисленным КД: редактировать нужно формулу или щит выше.
> > >     const row = dv.container.createDiv({ cls: "character-ac-result" });
> > >     row.style.display = "flex";
> > >     row.style.alignItems = "center";
> > >     row.style.flexWrap = "wrap";
> > >     row.style.gap = "10px";
> > >     row.style.margin = "8px 0";
> > >     const label = row.createEl("label", { text: "Итоговый КД:" });
> > >     label.style.fontWeight = "600";
> > >     const field = row.createEl("input", { type: "number" });
> > >     field.value = fmt(total);
> > >     field.readOnly = true;
> > >     field.setAttribute("aria-label", "Итоговый класс доспеха");
> > >     field.title = "Автоматически: формула КД + бонус надетого щита";
> > >     field.style.width = "82px";
> > >     field.style.fontWeight = "600";
> > >     field.style.textAlign = "center";
> > >     const breakdown = row.createEl("small", {
> > >         text: "Без щита: " + fmt(base) +
> > >             (shieldEquipped ? " · щит: +" + shieldBonus : " · щит снят")
> > >     });
> > >     breakdown.style.color = "var(--text-muted)";
> > > } catch (error) {
> > >     dv.paragraph("⚠️ **Ошибка формулы КД:** " +
> > >         String(error?.message ?? error).replace(/[<>&]/g, ""));
> > > }
> > > ```

const page = dv.current();
const asBoolean = value => value === true || value === "true" || value === 1;
const abilities = Array.from(page.character_abilities ?? [])
    .map((raw, index) => {
        const item = typeof raw === "string" ? { name: raw } : raw;
        if (!item || !item.name) return null;
        const max = Number(item.charges_max);
        const chargesMax = Number.isSafeInteger(max) && max > 0 ? max : null;
        const number = item.charges == null ? chargesMax : Number(item.charges);
        const charges = chargesMax === null ? null :
            (Number.isSafeInteger(number) ? Math.max(0, Math.min(chargesMax, number)) : chargesMax);
        const oldText = String(item.recharge ?? "").toLocaleLowerCase("ru");
        const hasShort = item.recharge_short !== undefined && item.recharge_short !== null;
        const hasLong = item.recharge_long !== undefined && item.recharge_long !== null;
        const rechargeShort = hasShort ? asBoolean(item.recharge_short)
            : /коротк|short/.test(oldText);
        const rechargeLong = hasLong ? asBoolean(item.recharge_long)
            : /длинн|продолжительн|long/.test(oldText) || (!oldText.trim() && chargesMax !== null);
        const rechargeNote = String(item.recharge_note ?? "").trim() ||
            (!/коротк|short|длинн|продолжительн|long/.test(oldText) ? oldText : "");
        return {
            id: String(item.id || "ability_legacy_" + index),
            name: String(item.name).trim(),
            description: String(item.description ?? ""),
            source: String(item.source ?? ""),
            activation: String(item.activation ?? ""),
            chargesMax, charges, rechargeShort, rechargeLong, rechargeNote
        };
    }).filter(item => item?.name);

const characterFile = app.vault.getAbstractFileByPath(page.file.path);
async function updateAbility(id, apply) {
    if (!characterFile || characterFile.extension !== "md") {
        throw new Error("Не удалось найти лист персонажа.");
    }
    let updated = null;
    await app.fileManager.processFrontMatter(characterFile, fm => {
        const list = Array.isArray(fm.character_abilities) ? [...fm.character_abilities] : [];
        const i = list.findIndex((raw, index) => {
            const candidate = raw && typeof raw === "object" ? String(raw.id || "ability_legacy_" + index)
                : "ability_legacy_" + index;
            return candidate === id;
        });
        if (i < 0) throw new Error("Способность больше не находится в списке.");
        const original = list[i];
        const item = typeof original === "string"
            ? { id, name: original } : { ...original, id: original.id || id };
        apply(item);
        updated = item;
        list[i] = item;
        fm.character_abilities = list;
    });
    return updated;
}

    dv.paragraph("_Карточки способностей пока не добавлены._");

        const title = card.createEl("summary", { text: ability.name });

        const content = card.createDiv();

        for (const [label, value] of [

            const line = content.createDiv();

            line.createEl("strong", { text: label + ": " });
            line.appendText(value);
        }

        if (ability.chargesMax !== null) {
            let chargeCount = ability.charges;
            const resource = content.createDiv();
            resource.style.display = "flex";
            resource.style.alignItems = "center";
            resource.style.gap = "8px";
            resource.style.flexWrap = "wrap";
            resource.style.margin = "8px 0";
            const meter = resource.createEl("strong", {
                text: "Заряды: " + chargeCount + " / " + ability.chargesMax
            });
            const use = resource.createEl("button", { text: "−1" });
            const refill = resource.createEl("button", { text: "+1" });
            use.setAttr("aria-label", "Потратить заряд способности " + ability.name);
            refill.setAttr("aria-label", "Вернуть один заряд способности " + ability.name);
            const refresh = () => {
                meter.setText("Заряды: " + chargeCount + " / " + ability.chargesMax);
                use.disabled = chargeCount <= 0;
                refill.disabled = chargeCount >= ability.chargesMax;
            };
            const change = async difference => {
                use.disabled = true;
                refill.disabled = true;
                try {
                    const result = await updateAbility(ability.id, entry => {
                        const max = Number(entry.charges_max);
                        if (!Number.isSafeInteger(max) || max < 1) {
                            throw new Error("У способности нет зарядов.");
                        }
                        const value = entry.charges == null ? max : Number(entry.charges);
                        if (!Number.isSafeInteger(value)) throw new Error("Некорректное число зарядов.");
                        entry.charges = Math.max(0, Math.min(max, value + difference));
                    });
                    chargeCount = Number(result.charges);
                    refresh();
                } catch (error) {
                    refresh();
                    new obsidian.Notice("Ошибка изменения зарядов: " + (error?.message ?? error));
                }
            };

            refill.addEventListener("click", () => change(1));
            refresh();

            recovery.style.alignItems = "center";

                box.checked = originalValue;
                wrap.appendText(label);
                box.addEventListener("change", async () => {

                    box.disabled = true;
                    try {
                        await updateAbility(ability.id, entry => { entry[key] = value; });
                    } catch (error) {
                        box.checked = !value;
                        new obsidian.Notice("Ошибка переключателя: " + (error?.message ?? error));
                    } finally {
                        box.disabled = false;
                    }
                });
            };
            toggle("Короткий отдых", "recharge_short", ability.rechargeShort);
            toggle("Длинный отдых", "recharge_long", ability.rechargeLong);
            if (ability.rechargeNote) {
                const other = content.createDiv();
                other.style.color = "var(--text-muted)";
                other.style.fontSize = "0.9em";
                other.appendText("Другая перезарядка: " + ability.rechargeNote);
            }
        }

        const desc = content.createDiv({
            text: ability.description.trim() || "Описание пока не добавлено."
        });
        desc.style.whiteSpace = "pre-wrap";
        desc.style.marginTop = "8px";
        desc.style.lineHeight = "1.5";
