const page = dv.current();
const asBoolean = value => value === true || value === "true" || value === 1;
const abilities = Array.from(page.character_abilities ?? [])
    .map((raw, index) => {
        const item = typeof raw === "string" ? { name: raw } : raw;
        if (!item || !item.name) return null;
        const max = Number(item.charges_max);
        const chargesMax = Number.isSafeInteger(max) && max > 0 ? max : null;
        const number = item.charges == null ? chargesMax : Number(item.charges);
        const charges = chargesMax === null ? null :
            (Number.isSafeInteger(number) ? Math.max(0, Math.min(chargesMax, number)) : chargesMax);
        const oldText = String(item.recharge ?? "").toLocaleLowerCase("ru");
        const hasShort = item.recharge_short !== undefined && item.recharge_short !== null;
        const hasLong = item.recharge_long !== undefined && item.recharge_long !== null;
        const rechargeShort = hasShort ? asBoolean(item.recharge_short)
            : /коротк|short/.test(oldText);
        const rechargeLong = hasLong ? asBoolean(item.recharge_long)
            : /длинн|продолжительн|long/.test(oldText) || (!oldText.trim() && chargesMax !== null);
        const rechargeNote = String(item.recharge_note ?? "").trim() ||
            (!/коротк|short|длинн|продолжительн|long/.test(oldText) ? oldText : "");
        return {
            id: String(item.id || "ability_legacy_" + index),
            name: String(item.name).trim(),
            description: String(item.description ?? ""),
            source: String(item.source ?? ""),
            activation: String(item.activation ?? ""),
            chargesMax, charges, rechargeShort, rechargeLong, rechargeNote
        };
    }).filter(item => item?.name);

const characterFile = app.vault.getAbstractFileByPath(page.file.path);
async function updateAbility(id, apply) {
    if (!characterFile || characterFile.extension !== "md") {
        throw new Error("Не удалось найти лист персонажа.");
    }
    let updated = null;
    await app.fileManager.processFrontMatter(characterFile, fm => {
        const list = Array.isArray(fm.character_abilities) ? [...fm.character_abilities] : [];
        const i = list.findIndex((raw, index) => {
            const candidate = raw && typeof raw === "object" ? String(raw.id || "ability_legacy_" + index)
                : "ability_legacy_" + index;
            return candidate === id;
        });
        if (i < 0) throw new Error("Способность больше не находится в списке.");
        const original = list[i];
        const item = typeof original === "string"
            ? { id, name: original } : { ...original, id: original.id || id };
        apply(item);
        updated = item;
        list[i] = item;
        fm.character_abilities = list;
    });
    return updated;
}

        ].filter(pair => pair[1])) {
            const line = content.createDiv();
            line.style.marginBottom = "4px";
            line.createEl("strong", { text: label + ": " });
            line.appendText(value);
        }

        if (ability.chargesMax !== null) {
            let chargeCount = ability.charges;
            const resource = content.createDiv();
            resource.style.display = "flex";
            resource.style.alignItems = "center";
            resource.style.gap = "8px";
            resource.style.flexWrap = "wrap";

            const meter = resource.createEl("strong", {

            });
            const use = resource.createEl("button", { text: "−1" });
            const refill = resource.createEl("button", { text: "+1" });
            use.setAttr("aria-label", "Потратить заряд способности " + ability.name);
            refill.setAttr("aria-label", "Вернуть один заряд способности " + ability.name);
            const refresh = () => {
                meter.setText("Заряды: " + chargeCount + " / " + ability.chargesMax);
                use.disabled = chargeCount <= 0;
                refill.disabled = chargeCount >= ability.chargesMax;
            };
            const change = async difference => {
                use.disabled = true;
                refill.disabled = true;
                try {
                    const result = await updateAbility(ability.id, entry => {
                        const max = Number(entry.charges_max);
                        if (!Number.isSafeInteger(max) || max < 1) {
                            throw new Error("У способности нет зарядов.");
                        }
                        const value = entry.charges == null ? max : Number(entry.charges);
                        if (!Number.isSafeInteger(value)) throw new Error("Некорректное число зарядов.");
                        entry.charges = Math.max(0, Math.min(max, value + difference));
                    });
                    chargeCount = Number(result.charges);
                    refresh();
                } catch (error) {
                    refresh();
                    new obsidian.Notice("Ошибка изменения зарядов: " + (error?.message ?? error));
                }
            };
            use.addEventListener("click", () => change(-1));
            refill.addEventListener("click", () => change(1));
            refresh();

            const recovery = content.createDiv();
            recovery.style.display = "flex";
            recovery.style.alignItems = "center";
            recovery.style.flexWrap = "wrap";
            recovery.style.gap = "10px";
            recovery.style.margin = "8px 0";
            const toggle = (label, key, originalValue) => {
                const wrap = recovery.createEl("label");
                wrap.style.display = "inline-flex";
                wrap.style.alignItems = "center";
                wrap.style.gap = "4px";
                const box = wrap.createEl("input", { type: "checkbox" });
                box.checked = originalValue;
                wrap.appendText(label);
                box.addEventListener("change", async () => {
                    const value = box.checked;
                    box.disabled = true;
                    try {
                        await updateAbility(ability.id, entry => { entry[key] = value; });
                    } catch (error) {
                        box.checked = !value;
                        new obsidian.Notice("Ошибка переключателя: " + (error?.message ?? error));
                    } finally {
                        box.disabled = false;
                    }
                });
            };
            toggle("Короткий отдых", "recharge_short", ability.rechargeShort);
            toggle("Длинный отдых", "recharge_long", ability.rechargeLong);
            if (ability.rechargeNote) {
                const other = content.createDiv();
                other.style.color = "var(--text-muted)";
                other.style.fontSize = "0.9em";
                other.appendText("Другая перезарядка: " + ability.rechargeNote);
            }
        }

        const desc = content.createDiv({
            text: ability.description.trim() || "Описание пока не добавлено."
        });
        desc.style.whiteSpace = "pre-wrap";
        desc.style.marginTop = "8px";
        desc.style.lineHeight = "1.5";
    }
}

_3-й уровень, умение линчевателя_

Вы можете выйти за свои физические и умственные пределы с чистой решимостью, на короткое время улучшив свои дедуктивные и боевые способности за счет ваших жизненных сил. Бонусным действием в свой ход вы можете потратить 1 Кость Хитов, чтобы заострить свое внимание наиболее грандиозных схемах или наиболее важных деталях. Когда вы входите в это обострённое состояние, прежде чем начнёте концентрироваться, бросьте Кость Хитов и вычтите выпавшее значение из общего количества ваших хитов. Этот урон невозможно никак уменьшить.

Пока вы концентрируетесь (как будто накладывая заклинание), вы получаете следующие преимущества:

- Вы получаете бонус к урону, который наносите своими атаками оружием, равный вашему бонусу мастерства.
- Всякий раз, когда существо совершает проверку Мудрости, Интеллекта или Харизмы, чтобы обнаружить ваше присутствие или определить ваше намерение, вы можете реакцией наложить помеху на этот бросок.

Эти эффекты длятся в количестве минут, равных выпавшему значению на Кости Хитов, и заканчиваются раньше, если вы потеряете концентрацию, станете недееспособным или умираете. У вас есть два использования этого умения, и вы восстанавливаете всё израсходованные использования после окончания продолжительного отдыха.

const page = dv.current();
const entries = Array.from(page.character_equipment ?? [])
    .map(x => typeof x === "string"
        ? {path:x,quantity:1,equipped:false,attuned:false}
        : {path:String(x?.path??""),quantity:Math.max(1,Math.floor(Number(x?.quantity)||1)),
           equipped:Boolean(x?.equipped),attuned:Boolean(x?.attuned)})
    .filter(x=>x.path.startsWith("Предметы/")&&x.path.endsWith(".md"));
const note=app.vault.getAbstractFileByPath(page.file.path);
const save=async(path,modify)=>{
    if(!note||note.extension!=="md") throw new Error("Лист персонажа не найден.");
    await app.fileManager.processFrontMatter(note,fm=>{
        const list=(Array.isArray(fm.character_equipment)?fm.character_equipment:[])
            .map(x=>typeof x==="string"
                ? {path:x,quantity:1,equipped:false,attuned:false}
                : {...x,path:String(x?.path??""),
                   quantity:Math.max(1,Math.floor(Number(x?.quantity)||1)),
                   equipped:Boolean(x?.equipped),attuned:Boolean(x?.attuned)});
        modify(list,path);
        fm.character_equipment=list;
    });
};
if(!entries.length){
    dv.paragraph("_У персонажа пока нет снаряжения._");
}else{
    const list=dv.container.createDiv({cls:"spellbook-list"});
    for(const entry of entries){
        const path=entry.path;
        const target=app.vault.getAbstractFileByPath(path);
        const fm=target?app.metadataCache.getFileCache(target)?.frontmatter??{}:{};
        let type=String(fm.item_type??"");
        let rarity=String(fm.item_rarity??"");
        if(target&&(!type||!rarity)){
            const body=await app.vault.cachedRead(target);
            if(!type)type=String(body.match(/^Тип:\s*(.+)$/m)?.[1]??"").trim();
            if(!rarity)rarity=String(body.match(/^Редкость:\s*(.+)$/m)?.[1]??"").trim();
        }
        const row=list.createDiv({cls:"spellbook-entry"});
        row.style.display="flex";
        row.style.alignItems="center";
        row.style.gap="8px";
        row.style.flexWrap="wrap";
        row.style.marginBottom="9px";
        const link=row.createEl("a",{text:target?.basename??path.split("/").pop().replace(/\.md$/,""),
            cls:"internal-link"});
        link.setAttr("href",path.replace(/\.md$/i,""));
        link.setAttr("data-href",path.replace(/\.md$/i,""));
        link.addEventListener("click",async e=>{
            e.preventDefault();
            if(!target){new obsidian.Notice("Карточка не найдена: "+path);return;}
            await app.workspace.getLeaf(e.ctrlKey||e.metaKey?"tab":false).openFile(target);
        });
        const details=[type,rarity].filter(Boolean);
        if(details.length){
            const info=row.createEl("small",{text:details.join(" · ")});
            info.style.color="var(--text-muted)";
        }
        const amount=row.createEl("input",{type:"number"});
        amount.value=String(entry.quantity);
        amount.min="1";
        amount.step="1";
        amount.style.width="58px";
        amount.title="Количество";
        amount.setAttr("aria-label","Количество "+(target?.basename??path));
        amount.addEventListener("change",async()=>{
            const amountValue=Number(amount.value);
            if(!Number.isSafeInteger(amountValue)||amountValue<1){
                amount.value=String(entry.quantity);
                new obsidian.Notice("Количество должно быть целым числом от 1.");
                return;
            }
            amount.disabled=true;
            try{
                await save(path,(list,p)=>{const x=list.find(x=>x.path===p);if(x)x.quantity=amountValue;});
                entry.quantity=amountValue;
            }catch(err){
                amount.value=String(entry.quantity);
                new obsidian.Notice("Ошибка количества: "+(err?.message??err));
            }finally{amount.disabled=false;}
        });
        const flag=(label,key)=>{
            const wrap=row.createEl("label");
            wrap.style.display="inline-flex";
            wrap.style.alignItems="center";
            wrap.style.gap="4px";
            const checkbox=wrap.createEl("input",{type:"checkbox"});
            checkbox.checked=entry[key];
            wrap.appendText(label);
            checkbox.addEventListener("change",async()=>{
                checkbox.disabled=true;
                try{
                    await save(path,(list,p)=>{const x=list.find(x=>x.path===p);if(x)x[key]=checkbox.checked;});
                    entry[key]=checkbox.checked;
                }catch(err){
                    checkbox.checked=entry[key];
                    new obsidian.Notice("Ошибка изменения предмета: "+(err?.message??err));
                }finally{checkbox.disabled=false;}
            });
        };
        flag("Надето","equipped");
        flag("Настроено","attuned");
        const remove=row.createEl("button",{text:"Убрать"});
        remove.setAttr("aria-label","Убрать предмет из инвентаря");
        remove.addEventListener("click",async()=>{
            remove.disabled=true;
            try{
                await save(path,(list,p)=>{
                    const i=list.findIndex(x=>x.path===p);
                    if(i>=0)list.splice(i,1);
                });
                row.remove();
            }catch(err){
                remove.disabled=false;
                new obsidian.Notice("Ошибка удаления: "+(err?.message??err));
            }
        });
    }
}
