```dataviewjs
const page = dv.current();
const entries = Array.from(page.character_equipment ?? [])
    .map(x => typeof x === "string"
        ? {path:x,quantity:1,equipped:false,attuned:false}
        : {path:String(x?.path??""),quantity:Math.max(1,Math.floor(Number(x?.quantity)||1)),
           equipped:Boolean(x?.equipped),attuned:Boolean(x?.attuned)})
    .filter(x=>x.path.startsWith("Предметы/")&&x.path.endsWith(".md"));
const note=app.vault.getAbstractFileByPath(page.file.path);
const save=async(path,modify)=>{
    if(!note||note.extension!=="md") throw new Error("Лист персонажа не найден.");
    await app.fileManager.processFrontMatter(note,fm=>{
        const list=(Array.isArray(fm.character_equipment)?fm.character_equipment:[])
            .map(x=>typeof x==="string"
                ? {path:x,quantity:1,equipped:false,attuned:false}
                : {...x,path:String(x?.path??""),
                   quantity:Math.max(1,Math.floor(Number(x?.quantity)||1)),
                   equipped:Boolean(x?.equipped),attuned:Boolean(x?.attuned)});
        modify(list,path);
        fm.character_equipment=list;
    });
};
if(!entries.length){
    dv.paragraph("_У персонажа пока нет снаряжения._");
}else{
    const list=dv.container.createDiv({cls:"spellbook-list"});
    for(const entry of entries){
        const path=entry.path;
        const target=app.vault.getAbstractFileByPath(path);
        const fm=target?app.metadataCache.getFileCache(target)?.frontmatter??{}:{};
        let type=String(fm.item_type??"");
        let rarity=String(fm.item_rarity??"");
        if(target&&(!type||!rarity)){
            const body=await app.vault.cachedRead(target);
            if(!type)type=String(body.match(/^Тип:\s*(.+)$/m)?.[1]??"").trim();
            if(!rarity)rarity=String(body.match(/^Редкость:\s*(.+)$/m)?.[1]??"").trim();
        }
        const row=list.createDiv({cls:"spellbook-entry"});
        row.style.display="flex";
        row.style.alignItems="center";
        row.style.gap="8px";
        row.style.flexWrap="wrap";
        row.style.marginBottom="9px";
        const link=row.createEl("a",{text:target?.basename??path.split("/").pop().replace(/\.md$/,""),
            cls:"internal-link"});
        link.setAttr("href",path.replace(/\.md$/i,""));
        link.setAttr("data-href",path.replace(/\.md$/i,""));
        link.addEventListener("click",async e=>{
            e.preventDefault();
            if(!target){new obsidian.Notice("Карточка не найдена: "+path);return;}
            await app.workspace.getLeaf(e.ctrlKey||e.metaKey?"tab":false).openFile(target);
        });
        const details=[type,rarity].filter(Boolean);
        if(details.length){
            const info=row.createEl("small",{text:details.join(" · ")});
            info.style.color="var(--text-muted)";
        }
        const amount=row.createEl("input",{type:"number"});
        amount.value=String(entry.quantity);
        amount.min="1";
        amount.step="1";
        amount.style.width="58px";
        amount.title="Количество";
        amount.setAttr("aria-label","Количество "+(target?.basename??path));
        amount.addEventListener("change",async()=>{
            const amountValue=Number(amount.value);
            if(!Number.isSafeInteger(amountValue)||amountValue<1){
                amount.value=String(entry.quantity);
                new obsidian.Notice("Количество должно быть целым числом от 1.");
                return;
            }
            amount.disabled=true;
            try{
                await save(path,(list,p)=>{const x=list.find(x=>x.path===p);if(x)x.quantity=amountValue;});
                entry.quantity=amountValue;
            }catch(err){
                amount.value=String(entry.quantity);
                new obsidian.Notice("Ошибка количества: "+(err?.message??err));
            }finally{amount.disabled=false;}
        });
        const flag=(label,key)=>{
            const wrap=row.createEl("label");
            wrap.style.display="inline-flex";
            wrap.style.alignItems="center";
            wrap.style.gap="4px";
            const checkbox=wrap.createEl("input",{type:"checkbox"});
            checkbox.checked=entry[key];
            wrap.appendText(label);
            checkbox.addEventListener("change",async()=>{
                checkbox.disabled=true;
                try{
                    await save(path,(list,p)=>{const x=list.find(x=>x.path===p);if(x)x[key]=checkbox.checked;});
                    entry[key]=checkbox.checked;
                }catch(err){
                    checkbox.checked=entry[key];
                    new obsidian.Notice("Ошибка изменения предмета: "+(err?.message??err));
                }finally{checkbox.disabled=false;}
            });
        };
        flag("Надето","equipped");
        flag("Настроено","attuned");
        const remove=row.createEl("button",{text:"Убрать"});
        remove.setAttr("aria-label","Убрать предмет из инвентаря");
        remove.addEventListener("click",async()=>{
            remove.disabled=true;
            try{
                await save(path,(list,p)=>{
                    const i=list.findIndex(x=>x.path===p);
                    if(i>=0)list.splice(i,1);
                });
                row.remove();
            }catch(err){
                remove.disabled=false;
                new obsidian.Notice("Ошибка удаления: "+(err?.message??err));
            }
        });
    }
}
```
