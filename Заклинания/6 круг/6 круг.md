const level = 6;
const folder = "Заклинания/6 круг";

    .where(p => String(p.spell_source ?? "").toLowerCase() === "homebrew" && Number(p.spell_level) === level)
    .sort(p => p.file.name, "asc");
if (!homebrew.length) {
    dv.paragraph("_Своих заклинаний пока нет._");

    dv.list(homebrew.map(p => p.file.link));
}
